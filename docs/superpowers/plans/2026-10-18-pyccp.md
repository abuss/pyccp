# pyccp v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build pyccp v1: an asyncio CCP runtime (agents + dataflow vars) over a Rust constraint core (store, propagators, DFS search) with a pluggable `Solver` trait, custom solver only.

**Architecture:** Python owns agent orchestration and the public API; Rust owns the store, propagators, and search. The Rust store is the single source of truth; Python `Var` is a handle to a store cell. Facts cross the boundary once per entailed fact via `SimpleQueue` + `call_soon_threadsafe`, never per propagation step.

**Tech Stack:** Rust 2021, pyo3 0.22 (extension-module), rayon 1; Python ≥ 3.11, pytest; maturin for the build.

**Spec:** `docs/superpowers/specs/2026-10-18-pyccp-design.md`

## Global Constraints

- Python ≥ 3.11; Rust edition 2021.
- Rust deps: `pyo3 = "0.30"` (feature `extension-module`; bumped from 0.22 because Python 3.14 cannot build 0.22), `rayon = "1"`. No tokio.
- Python dev deps: `pytest` only. Build with `maturin develop`.
- No Python callables in the hot loop; constraints cross the FFI as data tuples.
- Integer (`i64`) domains only.
- One failing check per unit; no fixtures, no frameworks beyond pytest and cargo test.
- Commit after every task.

```rust
#[derive(Debug, PartialEq)]
pub struct Conflict;

pub struct Store {
    cells: Vec<Vec<i64>>,
    trail: Vec<(usize, i64)>,
}

impl Store {
    pub fn new(domains: &[Vec<i64>]) -> Store {
        Store { cells: domains.to_vec(), trail: Vec::new() }
    }

    pub fn domain(&self, cell: usize) -> &[i64] { &self.cells[cell] }
    pub fn is_fixed(&self, cell: usize) -> bool { self.cells[cell].len() == 1 }
    pub fn value(&self, cell: usize) -> i64 { self.cells[cell][0] }
    pub fn first_open(&self) -> Option<usize> {
        (0..self.cells.len()).find(|&c| !self.is_fixed(c))
    }
    pub fn mark(&self) -> usize { self.trail.len() }
    pub fn trail_len(&self) -> usize { self.trail.len() }

    pub fn remove(&mut self, cell: usize, value: i64) -> Result<(), Conflict> {
        let d = &mut self.cells[cell];
        if let Some(pos) = d.iter().position(|&v| v == value) {
            d.remove(pos);
            self.trail.push((cell, value));
            if d.is_empty() { return Err(Conflict); }
        }
        Ok(())
    }

    pub fn commit(&mut self, cell: usize, value: i64) -> Result<(), Conflict> {
        if !self.cells[cell].contains(&value) { return Err(Conflict); }
        for v in self.cells[cell].clone() {
            if v != value { self.remove(cell, v)?; }
        }
        Ok(())
    }

    pub fn undo_to(&mut self, mark: usize) {
        for (cell, value) in self.trail.drain(mark..).rev() {
            self.cells[cell].push(value);
            self.cells[cell].sort();
        }
    }

    pub fn assignment(&self) -> Vec<i64> {
        self.cells.iter().map(|d| d[0]).collect()
    }
}
```

```rust
use crate::store::{Conflict, Store};

pub trait Propagator: Send + Sync {
    fn cells(&self) -> &[usize];
    fn propagate(&mut self, store: &mut Store) -> Result<(), Conflict>;
}

pub struct NeqLinear { cells: Vec<usize>, terms: Vec<(usize, i64)>, k: i64 }
impl NeqLinear {
    pub fn new(terms: Vec<(usize, i64)>, k: i64) -> NeqLinear {
        NeqLinear { cells: terms.iter().map(|t| t.0).collect(), terms, k }
    }
}
impl Propagator for NeqLinear {
    fn cells(&self) -> &[usize] { &self.cells }
    fn propagate(&mut self, store: &mut Store) -> Result<(), Conflict> {
        // ponytail: prunes only when all-but-one term is fixed; add bounds
        // reasoning when interval propagation is needed.
        let open: Vec<(usize, i64)> = self.terms.iter()
            .filter(|(c, _)| !store.is_fixed(*c)).copied().collect();
        let mut sum = 0;
        for (c, k) in &self.terms {
            if store.is_fixed(*c) { sum += k * store.value(*c); }
        }
        if open.is_empty() {
            return if sum == self.k { Err(Conflict) } else { Ok(()) };
        }
        if open.len() == 1 {
            let (c, coef) = open[0];
            let rem = self.k - sum;
            if rem % coef == 0 {
                let v = rem / coef;
                if store.domain(c).contains(&v) { store.remove(c, v)?; }
            }
        }
        Ok(())
    }
}

pub struct AllDiff { cells: Vec<usize> }
impl AllDiff {
    pub fn new(cells: Vec<usize>) -> AllDiff { AllDiff { cells } }
}
impl Propagator for AllDiff {
    fn cells(&self) -> &[usize] { &self.cells }
    fn propagate(&mut self, store: &mut Store) -> Result<(), Conflict> {
        for &i in &self.cells {
            if store.is_fixed(i) {
                let v = store.value(i);
                for &j in &self.cells {
                    if j != i && store.domain(j).contains(&v) { store.remove(j, v)?; }
                }
            }
        }
        Ok(())
    }
}

pub struct LinearEq { cells: Vec<usize>, terms: Vec<(usize, i64)>, rhs: i64 }
impl LinearEq {
    pub fn new(terms: Vec<(usize, i64)>, rhs: i64) -> LinearEq {
        LinearEq { cells: terms.iter().map(|t| t.0).collect(), terms, rhs }
    }
}
impl Propagator for LinearEq {
    fn cells(&self) -> &[usize] { &self.cells }
    fn propagate(&mut self, store: &mut Store) -> Result<(), Conflict> {
        // ponytail: propagates only when all-but-one term is fixed; full
        // interval arithmetic when bounds-based pruning is needed.
        let open: Vec<(usize, i64)> = self.terms.iter()
            .filter(|(c, _)| !store.is_fixed(*c)).copied().collect();
        if open.len() > 1 { return Ok(()); }
        let mut sum = self.rhs;
        for (c, k) in &self.terms {
            if store.is_fixed(*c) { sum -= k * store.value(*c); }
        }
        if open.len() == 1 {
            let (c, k) = open[0];
            if sum % k != 0 { return Err(Conflict); }
            let v = sum / k;
            for cur in store.domain(c).to_vec() {
                if cur != v { store.remove(c, cur)?; }
            }
        } else {
            let mut total = 0;
            for (c, k) in &self.terms { total += k * store.value(*c); }
            if total != self.rhs { return Err(Conflict); }
        }
        Ok(())
    }
}

pub struct NonZero { cells: [usize; 1] }
impl NonZero {
    pub fn new(i: usize) -> NonZero { NonZero { cells: [i] } }
}
impl Propagator for NonZero {
    fn cells(&self) -> &[usize] { &self.cells }
    fn propagate(&mut self, store: &mut Store) -> Result<(), Conflict> {
        store.remove(self.cells[0], 0)
    }
}

pub struct Registry { props: Vec<Box<dyn Propagator>> }
impl Registry {
    pub fn new() -> Registry { Registry { props: Vec::new() } }
    pub fn add(&mut self, p: Box<dyn Propagator>) { self.props.push(p); }

    pub fn fixpoint(&mut self, store: &mut Store) -> Result<(), Conflict> {
        loop {
            let before = store.trail_len();
            for p in &mut self.props { p.propagate(store)?; }
            if store.trail_len() == before { return Ok(()); }
        }
    }
}
```

```rust
use crate::propagators::{AllDiff, LinearEq, NeqLinear, NonZero, Propagator, Registry};
use crate::store::{Conflict, Store};

#[derive(Clone, Debug)]
pub enum Constraint {
    NeqLinear(Vec<(usize, i64)>, i64),
    AllDiff(Vec<usize>),
    LinearEq(Vec<(usize, i64)>, i64),
    NonZero(usize),
}

fn to_propagator(c: &Constraint) -> Box<dyn Propagator> {
    match c {
        Constraint::NeqLinear(t, k) => Box::new(NeqLinear::new(t.clone(), *k)),
        Constraint::AllDiff(v) => Box::new(AllDiff::new(v.clone())),
        Constraint::LinearEq(t, r) => Box::new(LinearEq::new(t.clone(), *r)),
        Constraint::NonZero(i) => Box::new(NonZero::new(*i)),
    }
}

pub trait Solver {
    fn post(&mut self, c: Constraint) -> Result<(), Conflict>;
    fn entails(&self, c: &Constraint) -> bool;
    fn solve(&mut self) -> Option<Vec<i64>>;
}

pub struct CustomSolver {
    domains: Vec<Vec<i64>>,
    constraints: Vec<Constraint>,
    store: Store,
    registry: Registry,
}

impl CustomSolver {
    pub fn new(domains: &[Vec<i64>]) -> CustomSolver {
        CustomSolver {
            domains: domains.to_vec(),
            constraints: Vec::new(),
            store: Store::new(domains),
            registry: Registry::new(),
        }
    }
}

impl Solver for CustomSolver {
    fn post(&mut self, c: Constraint) -> Result<(), Conflict> {
        self.constraints.push(c.clone());
        self.registry.add(to_propagator(&c));
        self.registry.fixpoint(&mut self.store)?;
        Ok(())
    }

    fn entails(&self, c: &Constraint) -> bool {
        // ponytail: entailment = all involved cells fixed and constraint holds;
        // add real entailment (bounds reasoning) when needed.
        let involved: Vec<usize> = match c {
            Constraint::NeqLinear(t, _) | Constraint::LinearEq(t, _) => t.iter().map(|x| x.0).collect(),
            Constraint::AllDiff(v) => v.clone(),
            Constraint::NonZero(i) => vec![*i],
        };
        if !involved.iter().all(|&c| self.store.is_fixed(c)) { return false; }
        let mut scratch = Store::new(&self.domains);
        for &c in &involved { scratch.commit(c, self.store.value(c)).unwrap(); }
        to_propagator(c).propagate(&mut scratch).is_ok()
    }

    fn solve(&mut self) -> Option<Vec<i64>> {
        self.dfs()
    }
}

impl CustomSolver {
    fn dfs(&mut self) -> Option<Vec<i64>> {
        let cell = match self.store.first_open() {
            Some(c) => c,
            None => return Some(self.store.assignment()),
        };
        let domain: Vec<i64> = self.store.domain(cell).to_vec();
        for &v in &domain {
            let m = self.store.mark();
            if self.store.commit(cell, v).is_err() { continue; }
            if self.registry.fixpoint(&mut self.store).is_err() {
                self.store.undo_to(m);
                continue;
            }
            if let Some(sol) = self.dfs() { return Some(sol); }
            self.store.undo_to(m);
        }
        None
    }
}
```

```rust
impl CustomSolver {
    pub fn solve_parallel(&mut self) -> Option<Vec<i64>> {
        let cell = self.store.first_open()?;
        let domain: Vec<i64> = self.store.domain(cell).to_vec();
        let (first, rest) = (domain[0], domain[1..].to_vec());

        // main branch on the calling thread
        let m = self.store.mark();
        if self.store.commit(cell, first).is_ok() {
            if self.registry.fixpoint(&mut self.store).is_ok() {
                if let Some(sol) = self.dfs() { return Some(sol); }
            }
        }
        self.store.undo_to(m);

        // remaining splits on rayon workers, first solution wins
        let result: std::sync::Arc<std::sync::Mutex<Option<Vec<i64>>>> =
            std::sync::Arc::new(std::sync::Mutex::new(None));
        let constraints = self.constraints.clone();
        let domains = self.domains.clone();
        rayon::scope(|scope| {
            for &v in &rest {
                let (c, d, r) = (cell, domains.clone(), result.clone());
                let cons = constraints.clone();
                scope.spawn(move |_| {
                    if r.lock().unwrap().is_some() { return; }
                    let mut sub = CustomSolver::new(&d);
                    for k in cons { sub.post(k).unwrap(); }
                    if sub.store.commit(c, v).is_ok() && sub.registry.fixpoint(&mut sub.store).is_ok() {
                        if let Some(sol) = sub.dfs() {
                            *r.lock().unwrap() = Some(sol);
                        }
                    }
                });
            }
        });
        std::sync::Arc::try_unwrap(result).ok().map(|m| m.into_inner().unwrap()).flatten()
    }
}
```

Note: `post` inside workers re-runs fixpoint; that's fine — workers are solver-private and never notify.

```rust
use pyo3::create_exception;
use pyo3::exceptions::PyValueError;
use pyo3::prelude::*;

use crate::solver::{Constraint, CustomSolver, Solver};
use crate::store::Conflict;

create_exception!(pyccp_core, ConflictError, PyException);

fn to_pyerr(c: Conflict) -> PyErr { ConflictError::new_err("constraint conflict") }

fn parse_constraint(py: Python, c: &Bound<'_, PyAny>) -> PyResult<Constraint> {
    let kind: String = c.get_item(0)?.extract()?;
    let args = c.get_item(1)?;
    match kind.as_str() {
        "neq" => {
            let i: usize = args.get_item(0)?.extract()?;
            let j: usize = args.get_item(1)?.extract()?;
            Ok(Constraint::NeqLinear(vec![(i, 1), (j, -1)], 0))
        }
        "all_diff" => Ok(Constraint::AllDiff(args.get_item(0)?.extract()?)),
        "nonzero" => Ok(Constraint::NonZero(args.get_item(0)?.extract()?)),
        "linear_eq" => {
            let terms: Vec<(usize, i64)> = args.get_item(0)?.extract()?;
            let rhs = args.get_item(1)?.extract()?;
            Ok(Constraint::LinearEq(terms, rhs))
        }
        other => Err(PyValueError::new_err(format!("unknown kind: {other}"))),
    }
}

#[pyclass]
pub struct PyStore {
    solver: CustomSolver,
}

#[pymethods]
impl PyStore {
    #[new]
    fn new(domains: Vec<Vec<i64>>) -> PyStore {
        PyStore { solver: CustomSolver::new(&domains) }
    }

    fn post(&mut self, py: Python, c: &Bound<'_, PyAny>) -> PyResult<()> {
        let c = parse_constraint(py, c)?;
        self.solver.post(c).map_err(to_pyerr)
    }

    fn solve(&mut self) -> Option<Vec<i64>> {
        self.solver.solve()
    }
}

#[pymodule]
fn pyccp_core(m: &Bound<'_, PyModule>) -> PyResult<()> {
    m.add_class::<PyStore>()?;
    m.add("Conflict", ConflictError::new())?;
    Ok(())
}
```

```rust
#[pyclass]
pub struct PyStore {
    solver: CustomSolver,
    queue: Py<PyAny>,
    loop_: Py<PyAny>,
    wake: Py<PyAny>,
    fixed_before: Vec<bool>,
}

#[pymethods]
impl PyStore {
    #[new]
    fn new(domains: Vec<Vec<i64>>, queue: Py<PyAny>, loop_: Py<PyAny>, wake: Py<PyAny>) -> PyStore {
        let n = domains.len();
        PyStore {
            solver: CustomSolver::new(&domains),
            queue, loop_, wake,
            fixed_before: vec![false; n],
        }
    }

    fn post(&mut self, py: Python, c: &Bound<'_, PyAny>) -> PyResult<()> {
        let c = parse_constraint(py, c)?;
        self.solver.post(c).map_err(to_pyerr)?;
        self.notify_newly_fixed(py)?;
        Ok(())
    }

    fn commit(&mut self, py: Python, cell: usize, value: i64) -> PyResult<()> {
        self.solver.commit(cell, value).map_err(to_pyerr)?;
        self.notify_newly_fixed(py)?;
        Ok(())
    }

    fn solve(&mut self, py: Python) -> Option<Vec<i64>> {
        let sol = self.solver.solve();
        // main branch only: report final fixed cells
        for (cell, &v) in sol.iter().enumerate() {
            if !self.fixed_before[cell] {
                self.notify(py, cell, Some(v)).ok();
            }
        }
        sol
    }
}

impl PyStore {
    fn notify(&self, py: Python, cell: usize, value: Option<i64>) -> PyResult<()> {
        let fact = match value {
            Some(v) => (cell, v).into_py(py),
            None => (cell, py.None()).into_py(py),
        };
        self.queue.bind(py).call_method1("put", (fact,))?;
        let set = self.wake.bind(py).getattr("set")?;
        self.loop_.bind(py).call_method1("call_soon_threadsafe", (set,))?;
        Ok(())
    }

    fn notify_newly_fixed(&mut self, py: Python) -> PyResult<()> {
        for cell in 0..self.fixed_before.len() {
            if !self.fixed_before[cell] && self.solver.is_fixed(cell) {
                self.fixed_before[cell] = true;
                self.notify(py, cell, Some(self.solver.value(cell)))?;
            }
        }
        Ok(())
    }
}
```

Add `is_fixed(cell) -> bool` and `value(cell) -> i64` and `commit(cell, value)` to the `Solver` trait (implement on `CustomSolver` by delegating to `store`), so `bindings.rs` can call them.

```python
import asyncio
from queue import SimpleQueue
import pyccp_core

Conflict = pyccp_core.Conflict


class Var:
    def __init__(self, store, name, cell):
        self.store, self.name, self.cell = store, name, cell
        self._value = None
        self._event = asyncio.Event()

    def is_fixed(self) -> bool:
        return self._event.is_set()

    async def commit(self, value: int) -> None:
        self.store._core.commit(self.cell, value)

    async def read(self) -> int:
        self.store._readers[self.cell] = asyncio.current_task()
        await self._event.wait()
        return self._value


class Store:
    def __init__(self, domains: list[list[int]]):
        self._loop = asyncio.get_running_loop()
        self._queue = SimpleQueue()
        self._wake = asyncio.Event()
        self._core = pyccp_core.Store(domains, self._queue, self._loop, self._wake)
        self._vars: dict[int, Var] = {}
        self._readers: dict[int, asyncio.Task] = {}

    def var(self, name: str, cell: int) -> Var:
        v = Var(self, name, cell)
        self._vars[cell] = v
        return v

    def post(self, c: tuple) -> None:
        self._core.post(c)

    def solve(self) -> list[int] | None:
        return self._core.solve()

    def _apply(self, cell: int, value) -> None:
        v = self._vars.get(cell)
        if v is None:
            return
        if value is None:  # undo
            v._value = None
            v._event.clear()
            t = self._readers.pop(cell, None)
            if t is not None and not t.done():
                t.cancel()
        else:
            v._value = value
            v._event.set()

    def drain(self) -> None:
        while not self._queue.empty():
            cell, value = self._queue.get()
            self._apply(cell, value)
```

```python
async def join(*agents):
    return await asyncio.gather(*[a() for a in agents])

async def choice(*agents):
    tasks = [asyncio.ensure_future(a()) for a in agents]
    done, pending = await asyncio.wait(tasks, return_when=asyncio.FIRST_COMPLETED)
    for t in pending:
        t.cancel()
    return done.pop().result()

async def at(var: Var) -> int:
    return await var.read()
```
